[[https://github.com/cloudflare/cloudflared][Cloudflared]] proxies traffic from the Cloudflare network to your origins. This daemon sits between Cloudflare network and your origin (e.g. a webserver). Cloudflare attracts client requests and sends them to you via this daemon, without requiring you to poke holes on your firewall

#+NAME: Deploy cloudflared via helm
#+begin_src tmux
# Set deployment environment
export environment=dev
export application=gitea
export owner=jamma

# Create directories if they don't already exist
mkdir --parent "/home/${USER}/Services/${application}"

# Clone charts repository
cd "/home/${USER}/Services/${application}" && git clone https://gitea.jamma.life/jmhbnz/charts.git

# Ensure the repository is up to date
cd charts/${application} && git pull

# Install the chart
helm upgrade --install ${owner}-${application}-cloudflared-${environment} . \
    --namespace ${owner}-${application}-${environment} \
    --create-namespace
#+end_src
